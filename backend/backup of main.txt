def set_esp32_emotion(emotion: str):
    """Sends a non-blocking request to the ESP32 to change the face emotion."""
    if not ESP32_IP: return
    try:
        requests.get(f"{ESP32_IP}/emotion?type={emotion}", timeout=0.8)
    except Exception as e:
        logger.debug(f"Could not reach ESP32 for emotion: {e}")

def set_esp32_eyes(r: int, g: int, b: int):
    """Sends a non-blocking request to the ESP32 to change the eye color."""
    if not ESP32_IP: return
    try:
        requests.get(f"{ESP32_IP}/eyes?r={r}&g={g}&b={b}", timeout=0.8)
    except Exception as e:
        logger.debug(f"Could not reach ESP32 for eyes: {e}")

def jarvis_speak(text: str):
    """Wrapper function to mute the mic while Jarvis speaks to prevent echo loops."""
    global IS_SPEAKING
    IS_SPEAKING = True
    try:
        get_current_tts().speak(text)
    except Exception as e:
        logger.error(f"TTS Error: {e}")
    finally:
        time.sleep(0.5) # Wait half a second for room echoes to fade
        IS_SPEAKING = False

# ==========================================
AUDIO_BUFFER = []
IS_RECORDING = False
SILENCE_START = None
SILENCE_THRESHOLD = 300
SILENCE_DURATION = 1.5   
PROCESSING_AUDIO = False

def process_robot_audio():
    global AUDIO_BUFFER, PROCESSING_AUDIO
    PROCESSING_AUDIO = True
    set_esp32_emotion("thinking")
    
    temp_wav = f"robot_mic_{uuid.uuid4().hex}.wav"
    try:
        raw_data = b"".join(AUDIO_BUFFER)
        AUDIO_BUFFER.clear()
        
        with wave.open(temp_wav, 'wb') as wav_file:
            wav_file.setnchannels(1)
            wav_file.setsampwidth(2) 
            wav_file.setframerate(16000)
            wav_file.writeframes(raw_data)
            
        with open(temp_wav, "rb") as f:
            transcription = groq_client.audio.transcriptions.create(
                file=("audio.wav", f.read()),
                model="whisper-large-v3",
                language=CURRENT_LANGUAGE  
            )
            
        user_text = transcription.text.strip()
        if user_text:
            logger.info(f"Robot Heard: {user_text}")
            threading.Thread(target=process_voice_and_chat, args=(user_text, None)).start()
        else:
            set_esp32_emotion("happy") 
            
    except Exception as e:
        logger.error(f"Robot Mic Processing Error: {e}")
        set_esp32_emotion("confused")
    finally:
        PROCESSING_AUDIO = False
        if os.path.exists(temp_wav):
            os.remove(temp_wav)

@a@app.post("/api/robot_mic")
async def robot_mic_stream(request: Request):
    global AUDIO_BUFFER, IS_RECORDING, SILENCE_START, PROCESSING_AUDIO, IS_SPEAKING
    
    # Drop audio chunks if Jarvis is talking so he doesn't hear himself
    if IS_SPEAKING:
        if IS_RECORDING:
            IS_RECORDING = False
            AUDIO_BUFFER.clear()
        return {"status": "muted_while_speaking"}

    if PROCESSING_AUDIO:
        return {"status": "busy"}

    chunk = await request.body()
    if not chunk:
        return {"status": "empty"}

    audio_data = np.frombuffer(chunk, dtype=np.int16)
    rms = np.sqrt(np.mean(np.square(audio_data, dtype=np.float32)))
    
    # --- NEW FAILSAFE: Force stop if recording goes on too long (approx 5 seconds) ---
    if IS_RECORDING and len(AUDIO_BUFFER) > 150: 
        logger.warning("Max recording time reached! Forcing process...")
        IS_RECORDING = False
        SILENCE_START = None
        threading.Thread(target=process_robot_audio, daemon=True).start()
        return {"status": "ok"}
    # ---------------------------------------------------------------------------------

    if rms > SILENCE_THRESHOLD:
        if not IS_RECORDING:
            IS_RECORDING = True
            set_esp32_emotion("listening")
            logger.info("Voice detected, recording...")
        SILENCE_START = None
        AUDIO_BUFFER.append(chunk)
    elif IS_RECORDING:
        AUDIO_BUFFER.append(chunk)
        if SILENCE_START is None:
            SILENCE_START = time.time()
        elif time.time() - SILENCE_START > SILENCE_DURATION:
            IS_RECORDING = False
            SILENCE_START = None
            threading.Thread(target=process_robot_audio, daemon=True).start()

    return {"status": "ok"}

def get_groq_vision_response(user_message: str) -> str:
    set_esp32_emotion("thinking")
    try:
        relevant_memory = memory_manager.query_memory(user_message, n_results=2)
        memory_context = f"\nRelevant Past Memories: {relevant_memory}" if relevant_memory else ""

        frame = vision_pipeline.get_latest_frame()
        if frame is None:
            fallback_ans = brain.think(f"{memory_context}\nUser says: {user_message}", language_mode=CURRENT_LANGUAGE)
            set_esp32_emotion("happy")
            return fallback_ans

        _, buffer = cv2.imencode('.jpg', frame)
        base64_image = base64.b64encode(buffer).decode('utf-8')

        lang_instruction = "Respond in Hindi." if CURRENT_LANGUAGE == "hi" else "Respond in English."

        completion = groq_client.chat.completions.create(
            model="qwen/qwen3.6-27b",
            messages=[
                {
                    "role": "user",
                    "content": [
                        {
                            "type": "text",
                            "text": f"You are JARVIS, an advanced AI robot assistant. {lang_instruction} {memory_context}\nAnswer the user prompt considering what you see in the image feed. User says: {user_message}"
                        },
                        {
                            "type": "image_url",
                            "image_url": {
                                "url": f"data:image/jpeg;base64,{base64_image}"
                            }
                        }
                    ]
                }
            ]
        )
        
        answer = completion.choices[0].message.content
        if "</think>" in answer:
            answer = answer.split("</think>")[-1].strip()
        set_esp32_emotion("happy")

        memory_manager.add_memory(
            doc_id=f"mem_{uuid.uuid4().hex[:8]}",
            text_content=f"User asked: {user_message}. Jarvis answered: {answer}",
            metadata={"language": CURRENT_LANGUAGE, "timestamp": time.time()}
        )

        return answer
    except Exception as e:
        logger.error(f"Groq Vision Error: {e}")
        fallback_ans = brain.think(user_message, language_mode=CURRENT_LANGUAGE)
        set_esp32_emotion("happy")
        return fallback_ans

def proactive_vision_daemon():
    global LAST_SPOKEN_TIME
    logger.info("Advanced Event-Driven Proactive Daemon online.")
    
    last_distance_state = "clear"
    
    while True:
        time.sleep(2)
        try:
            now = time.time()
            distance = motion.get_distance()
            
            if 0 < distance < 30 and last_distance_state == "clear":
                last_distance_state = "blocked"
                if (now - LAST_SPOKEN_TIME) > PROACTIVE_COOLDOWN:
                    LAST_SPOKEN_TIME = now
                    logger.info("Proactive Event: Obstacle proximity detected.")
                    set_esp32_emotion("surprised")
                    set_esp32_eyes(255, 100, 0)
                    jarvis_speak("Caution, sir. An obstacle has entered my immediate proximity.")
                    continue
            elif distance >= 30:
                last_distance_state = "clear"

            frame = vision_pipeline.get_latest_frame()
            if frame is not None and (now - LAST_SPOKEN_TIME) > PROACTIVE_COOLDOWN:
                LAST_SPOKEN_TIME = now
                set_esp32_emotion("happy")
                set_esp32_eyes(0, 255, 128)
                response = get_groq_vision_response("A person or environment change is detected. Greet them happily with a smile as JARVIS.")
                jarvis_speak(response)
        except Exception as e:
            pass

def autonomous_navigation_loop():
    global AUTONOMOUS_MODE
    logger.info("True Autonomous Wander & Patrol Loop online.")
    
    while True:
        if AUTONOMOUS_MODE:
            try:
                distance = motion.get_distance()
                if 0 < distance < 25:
                    set_esp32_emotion("angry")
                    set_esp32_eyes(255, 0, 0)
                    motion.execute_movement("backward", 1.0)
                    escape_dir = random.choice(["left", "right"])
                    motion.execute_movement(escape_dir, 0.8)
                    time.sleep(0.5)
                else:
                    set_esp32_emotion("happy")
                    set_esp32_eyes(0, 255, 204)
                    motion.execute_movement("forward", 0.8)
            except Exception as e:
                logger.error(f"Patrol Loop Error: {e}")
        time.sleep(0.3)

def wake_word_listener():
    logger.info("Wake Word Listener online. Say 'Hey Jarvis' to activate.")
    recognizer = sr.Recognizer()
    mic = sr.Microphone()
    
    with mic as source:
        recognizer.adjust_for_ambient_noise(source, duration=1)
    
    while True:
        try:
            if IS_SPEAKING:
                time.sleep(1)
                continue
                
            with mic as source:
                audio = recognizer.listen(source, phrase_time_limit=5)
            
            text = recognizer.recognize_google(audio).lower()
            logger.info(f"Heard Background: {text}")
            
            if "jarvis" in text:
                threading.Thread(target=process_voice_and_chat, args=(text, None)).start()
        except sr.UnknownValueError:
            pass 
        except Exception as e:
            time.sleep(1)

# VOICE COMMAND PROCESSOR (OVERRIDE, EYES, EMOTIONS, & MOVEMENT)
# --------------------------------------------------------------------------
def process_voice_and_chat(user_message: str, response_text: str = None):
    global AUTONOMOUS_MODE
    try:
        user_intent = user_message.lower()
        
        # 1. Voice Overrides for Autonomous Mode
        movement_words = ["explore", "forward", "backward", "back", "left", "right", "stop", "halt"]
        if any(word in user_intent for word in movement_words):
            if AUTONOMOUS_MODE:
                logger.info("Voice Override: Disabling Autonomous Mode for manual movement.")
                AUTONOMOUS_MODE = False
                set_esp32_emotion("happy") # Reset the face to normal if it was angry/patrolling
                set_esp32_eyes(0, 255, 204)
                
        if "patrol" in user_intent or "start autonomy" in user_intent or "enable autonomous" in user_intent:
            logger.info("Voice Command: Enabling Autonomous Mode.")
            AUTONOMOUS_MODE = True
            jarvis_speak("Initiating autonomous wander and patrol protocol, sir.")
            return

        # 2. Custom Voice Triggers for Eyes and Face
        if "red eyes" in user_intent or "eyes to red" in user_intent or "change face to red" in user_intent:
            set_esp32_eyes(255, 0, 0)
            motion.execute_eye_color(255, 0, 0)
            jarvis_speak("Setting optical array to red, sir.")
            return
        elif "blue eyes" in user_intent or "eyes to blue" in user_intent or "change face to blue" in user_intent:
            set_esp32_eyes(0, 150, 255)
            motion.execute_eye_color(0, 150, 255)
            jarvis_speak("Setting optical array to blue.")
            return
        elif "green eyes" in user_intent or "eyes to green" in user_intent or "change face to green" in user_intent:
            set_esp32_eyes(0, 255, 100)
            motion.execute_eye_color(0, 255, 100)
            jarvis_speak("Optical array set to green.")
            return
        elif "cyan eyes" in user_intent or "default eyes" in user_intent:
            set_esp32_eyes(0, 255, 204)
            motion.execute_eye_color(0, 255, 204)
            jarvis_speak("Resetting optical array to standard cyan.")
            return
        elif "angry face" in user_intent or "look angry" in user_intent:
            set_esp32_emotion("angry")
            set_esp32_eyes(255, 0, 0)
            jarvis_speak("Combat status displayed.")
            return
        elif "happy face" in user_intent or "smile" in user_intent:
            set_esp32_emotion("happy")
            set_esp32_eyes(0, 255, 204)
            jarvis_speak("Glad to assist you, sir.")
            return
        
        # 3. Direct Movement Execution
        if "forward" in user_intent or "move ahead" in user_intent:
            threading.Thread(target=motion.execute_movement, args=("forward", 2.0)).start()
        elif "backward" in user_intent or "back" in user_intent:
            threading.Thread(target=motion.execute_movement, args=("backward", 2.0)).start()
        elif "left" in user_intent:
            threading.Thread(target=motion.execute_movement, args=("left", 1.0)).start()
        elif "right" in user_intent:
            threading.Thread(target=motion.execute_movement, args=("right", 1.0)).start()
        elif "stop" in user_intent or "halt" in user_intent:
            threading.Thread(target=motion.execute_movement, args=("stop", 0.0)).start()
            set_esp32_emotion("happy") # Force reset face on stop command
            set_esp32_eyes(0, 255, 204)
            jarvis_speak("Stopping all motors.")
            return

        # 4. Multimodal AI Response & Speech Output
        if not response_text:
            response_text = get_groq_vision_response(user_message)
            
        sentences = re.split(r'(?<=[.!?])\s+', response_text)
        for sentence in sentences:
            if sentence.strip():
                jarvis_speak(sentence.strip())

    except Exception as e:
        logger.error(f"Error in Voice/Chat execution: {e}")

@app.post("/api/chat")
def api_chat(req: ChatRequest):
    global CURRENT_LANGUAGE
    CURRENT_LANGUAGE = req.language
    set_esp32_emotion("listening")
    
    response_text = get_groq_vision_response(req.message)
    threading.Thread(target=process_voice_and_chat, args=(req.message, response_text)).start()
    return {"status": "success", "response": response_text}

@app.post("/api/stt")
async def api_stt(audio: UploadFile = File(...)):
    set_esp32_emotion("listening")
    temp_audio = f"temp_stt_{uuid.uuid4().hex}.webm"
    try:
        with open(temp_audio, "wb") as f:
            f.write(await audio.read())
        
        with open(temp_audio, "rb") as f:
            transcription = groq_client.audio.transcriptions.create(
                file=("audio.webm", f.read()),
                model="whisper-large-v3",
                language=CURRENT_LANGUAGE
            )
            
        return {"status": "success", "text": transcription.text}
    except Exception as e:
        set_esp32_emotion("confused")
        logger.error(f"Groq STT Error: {e}")
        return {"status": "error", "text": ""}
    finally:
        if os.path.exists(temp_audio):
            os.remove(temp_audio)

def register_server_with_esp32():
    """Finds the laptop's local IP and tells the ESP32 where to send audio, retrying until successful."""
    if not ESP32_IP: return
    
    while True:
        try:
            s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
            s.connect(('10.255.255.255', 1))
            laptop_ip = s.getsockname()[0]
            s.close()
            
            url = f"{ESP32_IP}/set_server?ip={laptop_ip}"
            logger.info(f"Attempting to register Audio Server at {url} ...")
            
            response = requests.get(url, timeout=2)
            
            if response.status_code == 200:
                # REMOVED THE EMOJI HERE SO WINDOWS DOESN'T CRASH
                logger.info(f"[SUCCESS] Registered Audio Server with ESP32 at {laptop_ip}")
                break  # Exit the loop! We are connected.
            else:
                logger.warning(f"ESP32 returned status {response.status_code}. Retrying in 5 seconds...")
                
        except Exception as e:
            logger.error(f"ESP32 Network Error / Timeout. Is the robot on? Retrying in 5 seconds...")
        
        # Wait 5 seconds before trying again
        time.sleep(5)
