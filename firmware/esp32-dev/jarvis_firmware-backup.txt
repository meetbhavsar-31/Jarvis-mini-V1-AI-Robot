#include <WebServer.h>

// ==========================================
// AUTO-DISCOVERY VARIABLE
// ==========================================
String pythonServerUrl = ""; // Starts empty, Python will fill this in automatically!

// ==========================================
// ESP32 PIN DEFINITIONS 
// ==========================================
#define SDA_PIN 21 
#define SCL_PIN 22 
#define TFT_CS   5
#define TFT_RST  4
#define TFT_DC   2
#define TRIG_PIN 13
#define ECHO_PIN 12

// INMP441 I2S Pins
#define I2S_WS 33
#define I2S_SD 34
#define I2S_SCK 32

// PCF8574 Motor Pins
#define IN1 0 
#define IN2 1 
#define IN3 2 
#define IN4 3 

Adafruit_ST7735 tft = Adafruit_ST7735(TFT_CS, TFT_DC, TFT_RST);
WebServer server(80);
Adafruit_PCF8574 pcf;

// ==========================================
// STATE VARIABLES
// ==========================================
enum Emotion { THINKING, LISTENING, SURPRISE, SAD, HAPPY, ANGRY, SLEEPY, WINK, CONFUSED };
Emotion currentEmotion = HAPPY;
Emotion previousEmotion = THINKING; 

bool isSpeaking = false;
bool mouthOpen = false;
unsigned long lastAnimTime = 0;
unsigned long lastBlinkTime = 0;
bool isBlinking = false;
uint16_t currentEyeColor = 0x07FF; 

#define C_ORANGE 0xFD20
#define C_RED    0xF800
#define C_PURPLE 0x780F

// ==========================================
// AUDIO STREAMING TASK (RUNS ON CORE 0)
// ==========================================
#define I2S_PORT I2S_NUM_0
#define BUFFER_LEN 1024
int16_t sBuffer[BUFFER_LEN];

void i2s_install() {
  const i2s_config_t i2s_config = {
    .mode = i2s_mode_t(I2S_MODE_MASTER | I2S_MODE_RX),
    .sample_rate = 16000,
    .bits_per_sample = i2s_bits_per_sample_t(16),
    .channel_format = I2S_CHANNEL_FMT_ONLY_LEFT,
    .communication_format = i2s_comm_format_t(I2S_COMM_FORMAT_I2S | I2S_COMM_FORMAT_I2S_MSB),
    .intr_alloc_flags = 0,
    .dma_buf_count = 8,
    .dma_buf_len = BUFFER_LEN,
    .use_apll = false
  };
  i2s_driver_install(I2S_PORT, &i2s_config, 0, NULL);
}

void i2s_setpin() {
  const i2s_pin_config_t pin_config = {
    .bck_io_num = I2S_SCK,
    .ws_io_num = I2S_WS,
    .data_out_num = -1,
    .data_in_num = I2S_SD
  };
  i2s_set_pin(I2S_PORT, &pin_config);
}

void audioTask(void * pvParameters) {
  i2s_install();
  i2s_setpin();
  i2s_start(I2S_PORT);
  
  HTTPClient http;
  WiFiClient client;
  
  while(true) {
    if(WiFi.status() == WL_CONNECTED && pythonServerUrl != "") {
      size_t bytesIn = 0;
      esp_err_t result = i2s_read(I2S_PORT, &sBuffer, BUFFER_LEN * sizeof(int16_t), &bytesIn, portMAX_DELAY);
      
      if (result == ESP_OK && bytesIn > 0) {
        http.begin(client, pythonServerUrl.c_str());
        http.addHeader("Content-Type", "application/octet-stream");
        int httpResponseCode = http.POST((uint8_t*)sBuffer, bytesIn);
        http.end();
      }
    }
    vTaskDelay(10 / portTICK_PERIOD_MS); // Prevent watchdog timeout
  }
}

// ==========================================
// DRAWING ROUTINES (CORE 1)
// ==========================================
void drawListening() {
  tft.drawCircle(40, 50, 20, currentEyeColor); tft.drawCircle(40, 50, 10, currentEyeColor);
  tft.drawCircle(120, 50, 20, currentEyeColor); tft.drawCircle(120, 50, 10, currentEyeColor);
  tft.drawLine(65, 80, 95, 80, currentEyeColor); tft.drawLine(75, 75, 75, 85, currentEyeColor); tft.drawLine(85, 70, 85, 90, currentEyeColor);
}
void drawThinking() {
  for(int i=0; i<360; i+=30) {
    int x = 80 + 30 * cos(i * PI / 180.0); int y = 50 + 30 * sin(i * PI / 180.0); tft.fillCircle(x, y, 2, C_ORANGE);
  }
  tft.drawCircle(80, 50, 8, C_ORANGE); tft.drawLine(85, 55, 95, 65, C_ORANGE); 
}
void drawSurprise() {
  tft.fillCircle(40, 50, 20, ST7735_WHITE); tft.fillCircle(40, 50, 15, ST7735_BLACK);
  tft.fillCircle(120, 50, 20, ST7735_WHITE); tft.fillCircle(120, 50, 15, ST7735_BLACK);
  tft.drawTriangle(80, 75, 70, 95, 90, 95, C_ORANGE);
}
void drawSad() {
  tft.drawCircleHelper(40, 60, 20, 1, currentEyeColor); tft.drawCircleHelper(120, 60, 20, 2, currentEyeColor);
  tft.drawCircleHelper(80, 100, 25, 3, currentEyeColor);
}
void drawHappy() {
  tft.drawCircle(40, 45, 15, currentEyeColor); tft.drawCircle(40, 45, 5, currentEyeColor);
  tft.drawCircle(120, 45, 15, currentEyeColor); tft.drawCircle(120, 45, 5, currentEyeColor);
  tft.drawCircleHelper(80, 60, 30, 4, currentEyeColor); tft.drawCircleHelper(80, 61, 30, 4, currentEyeColor);
}
void drawAngry() {
  tft.drawRect(5, 5, 150, 118, C_RED); tft.drawLine(40, 40, 60, 60, C_RED); tft.drawLine(40, 45, 60, 65, C_RED);
  tft.drawLine(120, 40, 100, 60, C_RED); tft.drawLine(120, 45, 100, 65, C_RED);
}
void drawSleepy() {
  tft.fillRect(30, 60, 30, 4, currentEyeColor); tft.fillRect(100, 60, 30, 4, currentEyeColor);
  tft.drawRect(70, 90, 20, 10, ST7735_WHITE); tft.fillRect(72, 92, 5, 6, C_RED);
}
void drawWink() {
  tft.drawCircle(40, 50, 20, currentEyeColor); tft.drawCircle(40, 50, 10, currentEyeColor);
  tft.drawCircleHelper(120, 50, 20, 1, currentEyeColor); 
}
void drawConfused() {
  tft.setCursor(60, 30); tft.setTextColor(C_PURPLE); tft.setTextSize(6); tft.print("?");
  tft.fillTriangle(110, 60, 120, 70, 110, 80, ST7735_WHITE); tft.fillTriangle(110, 60, 100, 70, 110, 80, ST7735_WHITE);
}

void renderFace() {
  if (currentEmotion != previousEmotion) {
    tft.fillScreen(ST7735_BLACK);
    tft.setCursor(5, 5); tft.setTextColor(ST7735_WHITE); tft.setTextSize(1); tft.print("JARVIS MINI");
    tft.setCursor(120, 5); tft.setTextColor(currentEyeColor); tft.print("ONLINE");

    switch(currentEmotion) {
      case THINKING: drawThinking(); break;
      case LISTENING: drawListening(); break;
      case SURPRISE: drawSurprise(); break;
      case SAD: drawSad(); break;
      case HAPPY: drawHappy(); break;
      case ANGRY: drawAngry(); break;
      case SLEEPY: drawSleepy(); break;
      case WINK: drawWink(); break;
      case CONFUSED: drawConfused(); break;
    }
    previousEmotion = currentEmotion;
  }
}

void updateAnimations() {
  unsigned long currentMillis = millis();
  if (isSpeaking && (currentEmotion == HAPPY || currentEmotion == LISTENING)) {
    if (currentMillis - lastAnimTime > 150) {
      lastAnimTime = currentMillis; mouthOpen = !mouthOpen;
      tft.fillRect(70, 75, 20, 20, ST7735_BLACK); 
      if (mouthOpen) tft.fillRoundRect(70, 80, 20, 10, 3, currentEyeColor); 
      else tft.fillRect(70, 85, 20, 3, currentEyeColor); 
    }
  }
  if (!isBlinking && (currentMillis - lastBlinkTime > random(3000, 7000))) {
    isBlinking = true; lastBlinkTime = currentMillis;
    tft.fillRect(0, 30, 160, 40, ST7735_BLACK); tft.fillRect(30, 50, 30, 4, currentEyeColor); tft.fillRect(100, 50, 30, 4, currentEyeColor); 
  } 
  else if (isBlinking && (currentMillis - lastBlinkTime > 150)) {
    isBlinking = false; previousEmotion = THINKING; renderFace();
  }
}

// ==========================================
// WEB SERVER API
// ==========================================
void handleSetServer() {
  if (server.hasArg("ip")) {
    pythonServerUrl = "http://" + server.arg("ip") + ":8000/api/robot_mic";
    Serial.println("Audio target set to: " + pythonServerUrl);
    server.send(200, "text/plain", "Audio target set to: " + pythonServerUrl);
  } else {
    server.send(400, "text/plain", "Missing IP");
  }
}

void handleEmotion() {
  if (server.hasArg("type")) {
    String state = server.arg("type");
    if (state == "thinking") currentEmotion = THINKING; else if (state == "listening") currentEmotion = LISTENING;
    else if (state == "surprise") currentEmotion = SURPRISE; else if (state == "sad") currentEmotion = SAD;
    else if (state == "happy") currentEmotion = HAPPY; else if (state == "angry") currentEmotion = ANGRY;
    else if (state == "sleepy") currentEmotion = SLEEPY; else if (state == "wink") currentEmotion = WINK;
    else if (state == "confused") currentEmotion = CONFUSED;
    renderFace(); server.send(200, "text/plain", "Emotion updated");
  }
}

void handleTalk() {
  if (server.hasArg("state")) {
    isSpeaking = (server.arg("state") == "on");
    if (!isSpeaking) { tft.fillRect(70, 75, 20, 20, ST7735_BLACK); previousEmotion = THINKING; renderFace(); }
    server.send(200, "text/plain", isSpeaking ? "Talking: ON" : "Talking: OFF");
  }
}

void handleMove() {
  if (server.hasArg("dir")) {
    String dir = server.arg("dir");
    pcf.digitalWrite(IN1, LOW); pcf.digitalWrite(IN2, LOW); pcf.digitalWrite(IN3, LOW); pcf.digitalWrite(IN4, LOW);
    if (dir == "forward") { pcf.digitalWrite(IN1, HIGH); pcf.digitalWrite(IN3, HIGH); }
    else if (dir == "backward") { pcf.digitalWrite(IN2, HIGH); pcf.digitalWrite(IN4, HIGH); }
    else if (dir == "left") { pcf.digitalWrite(IN2, HIGH); pcf.digitalWrite(IN3, HIGH); }
    else if (dir == "right") { pcf.digitalWrite(IN1, HIGH); pcf.digitalWrite(IN4, HIGH); }
    server.send(200, "text/plain", "Moved " + dir);
  }
}

void handleDistance() {
  digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10); digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH, 30000); 
  server.send(200, "text/plain", String((duration * 0.0343) / 2.0));
}

void handleEyes() {
  if (server.hasArg("r") && server.hasArg("g") && server.hasArg("b")) {
    currentEyeColor = tft.color565(server.arg("r").toInt(), server.arg("g").toInt(), server.arg("b").toInt());
    previousEmotion = THINKING; renderFace(); server.send(200, "text/plain", "Eye color updated");
  }
}

// ==========================================
// SETUP & LOOP
// ==========================================
void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT); pinMode(ECHO_PIN, INPUT);

  tft.initR(INITR_BLACKTAB); tft.setRotation(1); tft.fillScreen(ST7735_BLACK);
  Wire.begin(SDA_PIN, SCL_PIN); pcf.begin(0x20, &Wire);
  for(int i=0; i<4; i++) { pcf.pinMode(i, OUTPUT); pcf.digitalWrite(i, LOW); }

  WiFiManager wifiManager; wifiManager.autoConnect("JARVIS_Setup"); 
  if (MDNS.begin("jarvis")) Serial.println("MDNS responder started");

  server.on("/set_server", HTTP_GET, handleSetServer); // Auto-Discovery Endpoint
  server.on("/emotion", HTTP_GET, handleEmotion);
  server.on("/talk", HTTP_GET, handleTalk); 
  server.on("/move", HTTP_GET, handleMove);
  server.on("/distance", HTTP_GET, handleDistance);
  server.on("/eyes", HTTP_GET, handleEyes);
  server.begin();

  renderFace();
  
  // Launch Audio Streaming Task on Core 0 (Leaves Core 1 for Screen/Web)
  xTaskCreatePinnedToCore(audioTask, "AudioTask", 10000, NULL, 1, NULL, 0);
}

void loop() {
  server.handleClient();
  updateAnimations(); 
}
