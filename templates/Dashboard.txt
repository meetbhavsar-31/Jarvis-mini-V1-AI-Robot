<script>
  // FIX (auth header missing): API_AUTH_TOKEN (if you set one in .env) is
  // rendered here by the server itself, from app.py's dashboard route --
  // so it always matches your .env automatically. If you never set
  // API_AUTH_TOKEN, this is just an empty string and every call below
  // skips the header (matches the backend's no-op-if-unset behavior).
  const API_TOKEN = "{{ api_token }}";

  function authHeaders(extra = {}) {
    const headers = { ...extra };
    if (API_TOKEN) headers['X-API-Token'] = API_TOKEN;
    return headers;
  }

  let isAutonomyActive = false;
  let currentLanguage = 'en';
  let previousRobotState = null;
  let recognition = null;
  let isListening = false;

  // FIX (intermittent keyboard controls): the old version reacted to single
  // keydown/keyup events only. Holding two keys together (e.g. W+D) and
  // releasing just one sent 'stop' even though the other was still held --
  // felt like keys randomly stopped working. Now we track the full set of
  // currently-held keys and recompute the correct direction from that set.
  // Also: each fetch is tagged with an increasing sequence number so an
  // out-of-order network response (e.g. a 'stop' request that happens to
  // arrive after a later 'forward' request) can never overwrite a more
  // recent command.
  const heldMovementKeys = new Set();
  let moveRequestSeq = 0;
  let lastSentDirection = null;

  function moveRobot(dir) {
    if (dir === lastSentDirection) return; // avoid spamming identical commands
    lastSentDirection = dir;
    const seq = ++moveRequestSeq;
    fetch(`/api/move?dir=${dir}&duration=0`, { headers: authHeaders() })
      .catch(() => {})
      .finally(() => {
        // If a newer command has already been sent by the time this
        // request's response comes back, don't let this stale one
        // "win" -- lastSentDirection has already moved on.
        if (seq !== moveRequestSeq) return;
      });
  }

  // Given everything currently held, decide the single direction to send.
  // Supports true diagonals (W+D, W+A, S+D, S+A) in addition to the
  // explicit Q/E diagonal keys, and correctly falls back to whichever
  // key is still held when one of a combo is released.
  function recomputeMovementFromHeldKeys() {
    const has = (k) => heldMovementKeys.has(k);
    let dir = 'stop';
    if (has('KeyW') && has('KeyD')) dir = 'forward_right';
    else if (has('KeyW') && has('KeyA')) dir = 'forward_left';
    else if (has('KeyS') && has('KeyD')) dir = 'backward_right';
    else if (has('KeyS') && has('KeyA')) dir = 'backward_left';
    else if (has('KeyW')) dir = 'forward';
    else if (has('KeyS')) dir = 'backward';
    else if (has('KeyA')) dir = 'left';
    else if (has('KeyD')) dir = 'right';
    else if (has('KeyQ')) dir = 'forward_left';
    else if (has('KeyE')) dir = 'forward_right';
    moveRobot(dir);
  }

  document.querySelectorAll('.dpad-btn').forEach(btn => {
    btn.addEventListener('mousedown', () => moveRobot(btn.dataset.direction));
    btn.addEventListener('touchstart', (e) => { e.preventDefault(); moveRobot(btn.dataset.direction); });
    btn.addEventListener('mouseup', () => moveRobot('stop'));
    btn.addEventListener('touchend', (e) => { e.preventDefault(); moveRobot('stop'); });
    btn.addEventListener('mouseleave', () => moveRobot('stop'));
  });

  const MOVEMENT_KEYS = ['KeyW', 'KeyS', 'KeyA', 'KeyD', 'KeyQ', 'KeyE'];

  window.addEventListener('keydown', (e) => {
    if (['input', 'textarea'].includes(document.activeElement.tagName.toLowerCase())) return;
    if (e.code === 'Space') { e.preventDefault(); heldMovementKeys.clear(); moveRobot('stop'); return; }
    if (MOVEMENT_KEYS.includes(e.code)) {
      e.preventDefault();
      if (!e.repeat) {
        heldMovementKeys.add(e.code);
        recomputeMovementFromHeldKeys();
      }
    }
  });

  window.addEventListener('keyup', (e) => {
    if (['input', 'textarea'].includes(document.activeElement.tagName.toLowerCase())) return;
    if (MOVEMENT_KEYS.includes(e.code)) {
      heldMovementKeys.delete(e.code);
      recomputeMovementFromHeldKeys(); // resumes whatever other key is still held, or stops if none
    }
  });

  // Safety net: if the browser tab loses focus while a key is physically
  // held (alt-tab, clicking another window), the keyup event never fires --
  // the robot would otherwise keep driving forever. Stop on blur.
  window.addEventListener('blur', () => {
    heldMovementKeys.clear();
    moveRobot('stop');
  });
    const host = window.location.host;
    fetch(`/api/cast-team?server=${host}`).catch(() => {
      fetch(`http://jarvis.local/team?server=${host}`).catch(() => {});
    });


  function castMemberToRobot(file, name, role) {
    const host = window.location.host;
    fetch(`/api/cast-member?server=${host}&file=${file}&name=${encodeURIComponent(name)}&role=${encodeURIComponent(role)}`).catch(() => {
      fetch(`http://jarvis.local/show_member?server=${host}&file=${file}&name=${encodeURIComponent(name)}&role=${encodeURIComponent(role)}`).catch(() => {});
    });
  }

  document.querySelectorAll('.dpad-btn').forEach(btn => {
    btn.addEventListener('mousedown', () => moveRobot(btn.dataset.direction));
    btn.addEventListener('touchstart', (e) => { e.preventDefault(); moveRobot(btn.dataset.direction); });
    btn.addEventListener('mouseup', () => moveRobot('stop'));
    btn.addEventListener('touchend', (e) => { e.preventDefault(); moveRobot('stop'); });
    btn.addEventListener('mouseleave', () => moveRobot('stop'));
  });

  window.addEventListener('keydown', (e) => {
    if (['input', 'textarea'].includes(document.activeElement.tagName.toLowerCase())) return;
    const keyMap = {
      'KeyW': 'forward', 'KeyS': 'backward', 'KeyA': 'left', 'KeyD': 'right',
      'KeyQ': 'forward_left', 'KeyE': 'forward_right',
      'Space': 'stop'
    };
    if (keyMap[e.code]) {
      e.preventDefault();
      moveRobot(keyMap[e.code]);
    }
  });

  window.addEventListener('keyup', (e) => {
    if (['input', 'textarea'].includes(document.activeElement.tagName.toLowerCase())) return;
    if (['KeyW', 'KeyS', 'KeyA', 'KeyD', 'KeyQ', 'KeyE'].includes(e.code)) {
      moveRobot('stop');
    }
  });

  if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    recognition = new SpeechRecognition();
    recognition.continuous = false;
    recognition.interimResults = false;

    recognition.onstart = () => {
      isListening = true;
      document.getElementById('micBtn').classList.add('listening');
    };

    recognition.onresult = (e) => {
      // FIX: Guard against "undefined" phantom texts from the mic
      if (e.results && e.results[0] && e.results[0][0]) {
        const transcript = e.results[0][0].transcript;
        if (transcript && transcript.trim() !== "undefined") {
          document.getElementById('dashboardChatInput').value = transcript;
          submitChat();
        }
      }
    };

    recognition.onerror = () => stopListening();
    recognition.onend = () => stopListening();
  }

  function toggleVoiceRecognition() {
    if (!recognition) {
      alert("Speech recognition is not supported in this browser.");
      return;
    }
    if (isListening) {
      recognition.stop();
    } else {
      recognition.lang = currentLanguage === 'hi' ? 'hi-IN' : 'en-US';
      try {
        recognition.start();
      } catch (err) {
        alert("Microphone access blocked! If you are using a mobile phone, browsers require you to access this dashboard via HTTPS (or localhost) to use the mic. Consider using 'ngrok' to get a secure HTTPS link.");
      }
    }
  }

  function stopListening() {
    isListening = false;
    document.getElementById('micBtn').classList.remove('listening');
  }

  function updateVolume(val) {
    document.getElementById('volLabel').innerText = val + "%";
    fetch('/api/volume', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({volume_percent: parseInt(val)}) });
  }

  function toggleAutonomyState() {
    isAutonomyActive = !isAutonomyActive;
    const btn = document.getElementById('autonomyToggleBtn');
    btn.innerText = isAutonomyActive ? 'Stop Autonomy' : 'Start Autonomy';
    btn.style.backgroundColor = isAutonomyActive ? '#da3633' : '';
    fetch('/api/control', { method: 'POST', headers: authHeaders({'Content-Type': 'application/json'}), body: JSON.stringify({command: isAutonomyActive ? 'autonomy_start' : 'autonomy_stop'}) });
  }

  function setLang(lang) {
    currentLanguage = lang;
    document.getElementById('langEnBtn').classList.toggle('active', lang === 'en');
    document.getElementById('langHiBtn').classList.toggle('active', lang === 'hi');
    fetch('/api/language', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({language: lang}) });

  function sendQuickChat(text) {
    document.getElementById('dashboardChatInput').value = text;
    submitChat();
  }

  function submitChat() {
    const input = document.getElementById('dashboardChatInput');
    const msg = input.value.trim();
    if (!msg) return;

    const chatBox = document.getElementById('dashboardChatMessages');
    chatBox.innerHTML += `<li style="margin-bottom:6px; color:white;"><b>You:</b> ${msg}</li>`;
    input.value = '';
    chatBox.scrollTop = chatBox.scrollHeight;

    fetch('/api/chat', {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({message: msg, language: currentLanguage})
    }).then(r => r.json()).then(res => {
      chatBox.innerHTML += `<li style="margin-bottom:6px; color:#58a6ff;"><b>Jarvis:</b> ${res.response}</li>`;
      chatBox.scrollTop = chatBox.scrollHeight;
    });
  }

  function takePhoto() {
    fetch('/api/control', {
      method: 'POST',
      headers: authHeaders({'Content-Type': 'application/json'}),
      body: JSON.stringify({command: 'screenshot'})
    });
    // Upload happens in a background thread server-side, so give it a
    // couple seconds before refreshing the gallery, then poll once more
    // shortly after in case the upload took a bit longer.
    setTimeout(fetchPhotoGallery, 2500);
    setTimeout(fetchPhotoGallery, 6000);
  }

  function fetchPhotoGallery() {
    fetch('/api/snapshots').then(r => r.json()).then(data => {
      const gallery = document.getElementById('photoGallery');
      if (data.snapshots && data.snapshots.length > 0) {
        gallery.innerHTML = data.snapshots.slice(0, 6).map(url => `
          <div style="min-width: 120px; background: #0d1117; padding: 6px; border-radius: 6px; border: 1px solid #30363d;">
            <img src="${url}" style="width: 100%; height: 90px; object-fit: cover; border-radius: 4px;" onclick="window.open('${url}', '_blank')" title="Click to open full size">
          </div>
        `).join('');
      }
    }).catch(() => {});
  }

      }
    }).catch(() => {});
  }, 2000);
